        // 开始分析
        async function startAnalysis() {
            const videoUrl = document.getElementById('videoUrl').value;
            const analysisType = document.getElementById('analysisType').value;
            const sampleSize = parseInt(document.getElementById('sampleSize').value);
            const dataSource = document.querySelector('input[name="dataSource"]:checked').value;
            
            // 验证输入
            if (!videoUrl.trim()) {
                alert('请输入B站视频链接或BV号');
                return;
            }
            
            // 提取BV号

            }

            loadingIndicator.style.display = 'block';

                el.style.display = 'none';
            });

                if (dataSource === 'real') {
                    // 尝试获取真实数据
                    analysisData = await fetchRealBilibiliData(bvid, analysisType, sampleSize);

            } catch (error) {

                resultsDiv.style.display = 'block';

                document.querySelectorAll('.results').forEach(el => {
                    el.style.display = 'block';
                });
            }
        }
        
        // 提取BV号
        function extractBvid(url) {
            // 尝试匹配BV号
            const bvMatch = url.match(/[Bb][Vv][0-9A-Za-z]+/);
            if (bvMatch) {
                return bvMatch[0];
            }
            
            // 尝试匹配视频ID
            const videoIdMatch = url.match(/video\/([0-9A-Za-z]+)/);
            if (videoIdMatch) {
                return videoIdMatch[1];
            }
            
            // 如果输入本身看起来像BV号
            if (/^[Bb][Vv][0-9A-Za-z]+$/.test(url)) {
                return url;
            }
            
            return null;
        }
        
        // 获取真实的B站数据（使用公开API）
        async function fetchRealBilibiliData(bvid, analysisType, sampleSize) {
            console.log(`正在获取B站数据: ${bvid}, 类型: ${analysisType}`);
            
            // 定义API端点（使用公开的代理API，但请注意这些API可能不稳定）
            const proxyUrl = 'https://api.bilibili.com';
            
            try {
                let videoInfo = null;
                let comments = [];
                let danmaku = [];
                
                // 1. 获取视频基本信息
                try {
                    const videoInfoUrl = `${proxyUrl}/x/web-interface/view?bvid=${bvid}`;
                    const videoResponse = await fetchWithTimeout(videoInfoUrl);
                    
                    if (!videoResponse.ok) {
                        throw new Error(`获取视频信息失败: ${videoResponse.status}`);
                    }
                    
                    const videoData = await videoResponse.json();
                    if (videoData.code !== 0) {
                        throw new Error(`B站API错误: ${videoData.message}`);
                    }
                    
                    videoInfo = videoData.data;
                    console.log('视频信息:', videoInfo);
                } catch (error) {
                    console.warn('获取视频信息失败，使用默认信息:', error);
                    videoInfo = {
                        title: `B站视频 ${bvid}`,
                        owner: { name: "UP主" },
                        stat: { view: 0, danmaku: 0, reply: 0 }
                    };
                }
                
                // 2. 获取评论数据
                if (analysisType !== 'danmaku') {
                    try {
                        const commentsUrl = `${proxyUrl}/x/v2/reply?type=1&oid=${videoInfo.aid || 0}&sort=2&ps=50`;
                        const commentsResponse = await fetchWithTimeout(commentsUrl);
                        
                        if (commentsResponse.ok) {
                            const commentsData = await commentsResponse.json();
                            if (commentsData.code === 0) {
                                comments = extractComments(commentsData.data.replies || [], sampleSize);
                            }
                        }
                    } catch (error) {
                        console.warn('获取评论失败，使用模拟评论:', error);
                    }
                    
                    // 如果没有获取到评论，使用模拟数据
                    if (comments.length === 0) {
                        comments = generateMockComments(sampleSize);
                    }
                }
                
                // 3. 获取弹幕数据（使用第三方API，因为官方API有CORS限制）
                if (analysisType !== 'comments') {
                    try {
                        // 使用一个公开的弹幕API（这个API可能不稳定）

        // 带超时的fetch
        async function fetchWithTimeout(url, timeout = 10000) {
            const controller = new AbortController();
            const timeoutId = setTimeout(() => controller.abort(), timeout);
            
            try {
                const response = await fetch(url, {
                    signal: controller.signal,
                    mode: 'cors',
                    headers: {
                        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
                    }
                });
                clearTimeout(timeoutId);
                return response;
            } catch (error) {
                clearTimeout(timeoutId);
                throw error;
            }
        }
        
        // 提取评论数据
        function extractComments(replies, maxCount) {
            const comments = [];
            const count = Math.min(maxCount, replies.length);
            
            for (let i = 0; i < count; i++) {
                const reply = replies[i];
                comments.push({
                    text: reply.content?.message || '无内容',
                    sentiment: "neutral", // 将在后续分析中确定
                    likes: reply.like || 0,
                    time: new Date((reply.ctime || 0) * 1000).toLocaleTimeString()
                });
            }
            
            return comments;
        }
        
        // 解析弹幕XML
        function parseDanmakuXML(xmlText, maxCount) {
            const danmaku = [];
            const parser = new DOMParser();
            
            try {
                const xmlDoc = parser.parseFromString(xmlText, "text/xml");
                const dElements = xmlDoc.getElementsByTagName('d');
                const count = Math.min(maxCount, dElements.length);
                
                for (let i = 0; i < count; i++) {
                    const d = dElements[i];
                    const text = d.textContent || '';
                    const p = d.getAttribute('p') || '';
                    const params = p.split(',');
                    
                    if (params.length >= 8) {
                        const time = parseFloat(params[0]) || 0;
                        const minutes = Math.floor(time / 60);
                        const seconds = Math.floor(time % 60);
                        const timeStr = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
                        
                        danmaku.push({
                            text: text,
                            time: timeStr,
                            sentiment: "neutral",
                            type: parseInt(params[1]) || 1
                        });
                    }
                }
            } catch (error) {
                console.warn('解析弹幕XML失败:', error);
            }
            
            return danmaku;
        }

        // 情感分析
        function analyzeSentiment(items) {
            const positiveWords = ["好", "棒", "赞", "精彩", "优秀", "厉害", "喜欢", "爱", "支持", "感谢", "感动", "有用", "不错", "可以", "牛逼", "666", "awesome", "great", "nice", "perfect"];
            const negativeWords = ["差", "烂", "垃圾", "无聊", "失望", "糟糕", "讨厌", "反感", "恶心", "弃坑", "取关", "广告", "水", "抄袭", "垃圾", "不好", "不行", "无语", "糟糕", "bad"];
            
            let positiveCount = 0;
            let negativeCount = 0;
            let neutralCount = 0;
            let totalScore = 0;
            
            items.forEach(item => {
                const text = item.text.toLowerCase();
                let score = 0;
                
                // 检查积极词汇
                positiveWords.forEach(word => {
                    if (text.includes(word)) score += 1;
                });
                
                // 检查消极词汇
                negativeWords.forEach(word => {
                    if (text.includes(word)) score -= 1;
                });
                
                // 确定情感
                if (score > 0) {
                    positiveCount++;
                    item.sentiment = "positive";
                } else if (score < 0) {
                    negativeCount++;
                    item.sentiment = "negative";
                } else {
                    neutralCount++;
                    item.sentiment = "neutral";
                }
                
                totalScore += score;
            });
            
            const total = items.length || 1;
            const avgScore = (totalScore / total).toFixed(2);
            
            return {
                positive: Math.round((positiveCount / total) * 100),
                negative: Math.round((negativeCount / total) * 100),
                neutral: Math.round((neutralCount / total) * 100),
                avgScore: avgScore
            };
        }
        
        // 提取关键词
        function extractKeywords(items) {
            const stopWords = ["的", "了", "在", "是", "我", "有", "和", "就", "不", "人", "都", "一", "一个", "上", "也", "很", "到", "说", "要", "去", "你", "会", "着", "没有", "看", "好", "自己", "这"];
            const wordCount = {};
            
            items.forEach(item => {
                const text = item.text;
                // 简单的中文分词（按字符分割）
                const words = text.split('');
                
                words.forEach(word => {
                    if (word.trim() && !stopWords.includes(word) && word.length === 1) {
                        wordCount[word] = (wordCount[word] || 0) + 1;
                    }
                });
            });
            
            // 取出现频率最高的20个词
            const sortedWords = Object.entries(wordCount)
                .sort((a, b) => b[1] - a[1])
                .slice(0, 20);
            
            const keywords = {};
            sortedWords.forEach(([word, count]) => {
                keywords[word] = count;
            });
            
            return keywords;
        }

            const keywords = Object.keys(analysisData.keywords);
            const frequencies = Object.values(analysisData.keywords);
            
            // 计算最大频率用于缩放
            const maxFreq = Math.max(...frequencies);
            
            keywords.forEach((keyword, index) => {
                const freq = frequencies[index];
                // 根据频率计算字体大小 (16px - 48px)
                const fontSize = 16 + (freq / maxFreq) * 32;
                
                // 根据频率计算颜色深浅
                const colorIntensity = Math.floor(100 + (freq / maxFreq) * 100);
                const color = `rgb(0, ${Math.floor(161 - (freq / maxFreq) * 50)}, ${214})`;
                
                const wordSpan = document.createElement('span');
                wordSpan.textContent = keyword;
                wordSpan.style.fontSize = `${fontSize}px`;
                wordSpan.style.color = color;
                wordSpan.style.margin = '5px 8px';
                wordSpan.style.padding = '2px 5px';
                wordSpan.style.cursor = 'pointer';
                wordSpan.style.transition = 'transform 0.2s';

                wordSpan.addEventListener('mouseenter', function() {

                });
                
                wordSpan.addEventListener('mouseleave', function() {
                    this.style.transform = 'scale(1)';
                    this.style.fontWeight = 'normal';
                });

        function exportReport() {

                return;
            }

            const blob = new Blob([reportText], { type: 'text/plain;charset=utf-8' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `B站舆情分析报告_${analysisData.videoId}_${new Date().toISOString().slice(0,10)}.txt`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
            
            alert('报告已导出为文本文件');
        }
